# bootctl manages an EFI system partition and boot loader variables; every VMM here is
# handed the kernel, so there is neither (packages/systemd/build.sh).
absent usr/bin/bootctl
# systemd's EFI stub and boot loader are built for the unified kernel image, which carries
# the stub outside the image (image/build-rootfs.sh); nothing in the image boots through
# either.
absent usr/lib/systemd/boot
# What -Dbootloader brings besides the stub (packages/systemd/build.sh): boot counting and
# the services and socket that run bootctl.
absent usr/lib/systemd/systemd-bless-boot
absent usr/lib/systemd/system-generators/systemd-bless-boot-generator
absent usr/lib/systemd/system/systemd-boot-random-seed.service
absent usr/lib/systemd/system/systemd-bootctl.socket

# The rest of what packages/systemd/build.sh drops with bootctl: boot counting, the
# services and sockets that would run it, and the stub's tmpfiles line.
absent usr/lib/systemd/systemd-boot-check-no-failures
absent usr/lib/systemd/system/systemd-bless-boot.service
absent usr/lib/systemd/system/systemd-boot-check-no-failures.service
absent usr/lib/systemd/system/sysinit.target.wants/systemd-boot-random-seed.service
absent usr/lib/systemd/system/systemd-boot-clear-sysfail.service
absent usr/lib/systemd/system/systemd-boot-update.service
absent usr/lib/systemd/system/systemd-bootctl@.service
absent usr/lib/systemd/system/sockets.target.wants/systemd-bootctl.socket
absent usr/lib/systemd/system/systemd-sysupdate-notify-bootctl@.service
absent usr/lib/systemd/system/systemd-sysupdate-notify-bootctl.socket
absent usr/lib/tmpfiles.d/20-systemd-stub.conf
